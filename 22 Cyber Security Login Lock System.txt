22 Cyber Security Login Lock System (While Loop)
#include <stdio.h>
int main() {
    int n, i = 0, x;
    int failed = 0, streak = 0, lockAt = -1;

    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &x);
        if (x == 0) {
            failed++;
            streak++;
            if (streak == 3 && lockAt == -1)
                lockAt = i + 1;
        } else {
            streak = 0;
        }
        i++;
    }

    if (lockAt == -1)
        printf("Lock Triggered At Attempt: Not Locked\n");
    else
        printf("Lock Triggered At Attempt: %d\n", lockAt);

    printf("Total Failed Attempts: %d", failed);
    return 0;
}
