 Largest of Three Numbers

#include <stdio.h>

int main() {

    int a=10;

    int b=15;

    int c=25;

    scanf("%d %d %d",&a,&b,&c);

    if(a>=b&& a>=c){

        printf("%d is largest",a);

        }else if(b>=a && b>=c){

            printf("%d is largest",b);

        }else{

            printf("invalid number");

        }

    return 0;

}
