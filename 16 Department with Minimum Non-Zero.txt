16 Department with Minimum Non-Zero Total
#include <stdio.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    int minSum = 1e9, idx = -1;

    for(int i=0;i<R;i++){
        int sum = 0;
        for(int j=0;j<C;j++){
            scanf("%d",&a[i][j]);
            sum += a[i][j];
        }
        if(sum!=0 && sum < minSum){
            minSum = sum;
            idx = i;
        }
    }
    printf("%d", idx);
return 0;
}
