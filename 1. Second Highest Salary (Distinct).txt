1. Second Highest Salary (Distinct)
#include <stdio.h>
int main() {
    int n, a[100];
    scanf("%d", &n);
    for(int i=0;i<n;i++) scanf("%d",&a[i]);
    int max = INT_MIN, second = INT_MIN;
    for(int i=0;i<n;i++){
        if(a[i] > max){
            second = max;
            max = a[i];
        } else if(a[i] < max && a[i] > second){
            second = a[i];
        }
    }
    if(second == INT_MIN)
        printf("No second highest");
    else
        printf("%d", second);
     return 0;
}
