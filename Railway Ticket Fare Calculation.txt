Railway Ticket Fare Calculation

#include<stdio.h>

int main(){

    int ticket;

    scanf("%d",&ticket);

    int age;

    scanf("%d",&age);

    switch(ticket){

        case 1:

        if(age<=12){

            int amo=(300-(300*0.50));

            printf("%d",amo);

        }else if (age>=63){

            int amo=(300-(300*0.33)-1);

            printf("%d",amo);

        }else{

            printf("300");

        }

        break;

        case 2:

          if(age<12){

              int amo = (1000-(1000*0.50));

              printf("%d",amo);

          }else{

              printf("1000");

          }

}

    return 0;

}
