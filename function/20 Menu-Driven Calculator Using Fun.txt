20 Menu-Driven Calculator Using Functions
#include <stdio.h>

float add(float a, float b) { return a + b; }
float sub(float a, float b) { return a - b; }
float mul(float a, float b) { return a * b; }
float divi(float a, float b) { return a / b; }

int main() {
    float a, b;
    char choice;
    scanf("%f %f %c", &a, &b, &choice);

    switch (choice) {
        case '+': printf("%.0f", add(a, b)); break;
        case '-': printf("%.0f", sub(a, b)); break;
        case '*': printf("%.0f", mul(a, b)); break;
        case '/': printf("%.2f", divi(a, b)); break;
        default: printf("Invalid");
    }
    return 0;
}
