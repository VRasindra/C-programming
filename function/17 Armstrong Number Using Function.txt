17 Armstrong Number Using Function
#include <stdio.h>
#include <math.h>

int countDigits(int n) {
    int count = 0;
    while (n != 0) {
        count++;
        n /= 10;
    }
    return count;
}

int isArmstrong(int n) {
    int original = n, sum = 0, digits = countDigits(n);
    while (n != 0) {
        int rem = n % 10;
        sum += pow(rem, digits);
        n /= 10;
    }
    return sum == original;
}

int main() {
    int N;
    scanf("%d", &N);
    if (isArmstrong(N))
        printf("Armstrong");
    else
        printf("Not Armstrong");
    return 0;
}
