21 Department with Longest Even–Odd Alternating Pattern
#include <stdio.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    int maxLen = 0, idx = 0;

    for(int i=0;i<R;i++){
        int len = 1;
        for(int j=1;j<C;j++){
            if((a[i][j]%2)!=(a[i][j-1]%2))
                len++;
            else break;
        }
        if(len > maxLen){
            maxLen = len;
            idx = i;
        }
    }
    printf("%d", idx);
return 0;
}
