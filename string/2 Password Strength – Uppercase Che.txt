2 Password Strength – Uppercase Check
#include <stdio.h>

int main() {
    char s[100];
    int i, found = 0;

    scanf("%s", s);

    for (i = 0; s[i] != '\0'; i++) {
        if (s[i] >= 'A' && s[i] <= 'Z') {
            found = 1;
            break;
        }
    }

    if (found)
        printf("Valid");
    else
        printf("Invalid");

    return 0;
}
