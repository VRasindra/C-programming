12 Special Character Counter in Passwords
#include <stdio.h>

int main() {
    char s[100];
    int i, count = 0;

    scanf("%s", s);

    for (i = 0; s[i] != '\0'; i++) {
        if (!((s[i]>='A'&&s[i]<='Z') ||
              (s[i]>='a'&&s[i]<='z') ||
              (s[i]>='0'&&s[i]<='9'))) {
            count++;
        }
    }

    printf("%d", count);
    return 0;
}
