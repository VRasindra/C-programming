4 Factory Machine Failure Streak Analyzer (While Loop)
#include <stdio.h>
int main() {
    int n, i = 0, x;
    int streak = 0, maxStreak = 0;

    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &x);
        if (x == 0) {
            streak++;
            if (streak > maxStreak)
                maxStreak = streak;
        } else {
            streak = 0;
        }
        i++;
    }
    printf("Longest Failure Streak: %d", maxStreak);
    return 0;
}
