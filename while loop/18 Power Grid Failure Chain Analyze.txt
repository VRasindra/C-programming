18 Power Grid Failure Chain Analyzer (While Loop)
#include <stdio.h>
int main() {
    int cap, n, i = 0, x;
    int safe = 0, fail = 0;

    scanf("%d", &cap);
    scanf("%d", &n);

    while (i < n) {
        scanf("%d", &x);
        if (x > cap)
            fail++;
        else
            safe++;
        i++;
    }
    printf("Safe Hours: %d\n", safe);
    printf("Failure Count: %d", fail);
    return 0;
}
