13 Mobile Data Pack Exhaustion Predictor (While Loop)
#include <stdio.h>
int main() {
    int data, n, i = 0, x;
    int exhaustedDay = -1;

    scanf("%d", &data);
    scanf("%d", &n);

    while (i < n) {
        scanf("%d", &x);
        data -= x;
        if (data <= 0 && exhaustedDay == -1) {
            exhaustedDay = i + 1;
            break;
        }
        i++;
    }

    if (exhaustedDay == -1) {
        printf("Exhausted Day: Not Exhausted\n");
        printf("Overused Data: 0");
    } else {
        printf("Exhausted Day: %d\n", exhaustedDay);
        printf("Overused Data: %d", data < 0 ? -data : 0);
    }
    return 0;
}
