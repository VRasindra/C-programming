15 Employee Overtime Risk Analyzer (While Loop)
#include <stdio.h>
int main() {
    int n, i = 0, x;
    int total = 0, burnout = 0;

    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &x);
        total += x;
        if (x > 4)
            burnout++;
        i++;
    }
    printf("Total Overtime: %d\n", total);
    printf("Burnout Days: %d", burnout);
    return 0;
}
