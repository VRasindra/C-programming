19 Hospital Emergency Queue Overflow System (While Loop)
#include <stdio.h>
int main() {
    int cap, n, i = 0, x;
    int treated = 0, rejected = 0;

    scanf("%d", &cap);
    scanf("%d", &n);

    while (i < n) {
        scanf("%d", &x);
        if (treated + x <= cap)
            treated += x;
        else {
            rejected += (treated + x) - cap;
            treated = cap;
        }
        i++;
    }
    printf("Treated Patients: %d\n", treated);
    printf("Rejected Patients: %d", rejected);
    return 0;
}
