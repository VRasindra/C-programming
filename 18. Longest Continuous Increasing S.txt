18. Longest Continuous Increasing Subarray
#include <stdio.h>
int main() {
    int n,a[100];
    scanf("%d",&n);
    for(int i=0;i<n;i++) scanf("%d",&a[i]);

    int maxLen=1,curr=1;
    for(int i=1;i<n;i++){
        if(a[i]>a[i-1]) curr++;
        else curr=1;
        if(curr>maxLen) maxLen=curr;
    }
    printf("%d",maxLen);
return 0;
}
