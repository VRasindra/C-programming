7 Hospital Bed Occupancy Threshold Monitor (While Loop)
#include <stdio.h>
int main() {
    int cap, n, i = 0, x;
    int occ = 0, critical = 0;

    scanf("%d", &cap);
    scanf("%d", &n);

    while (i < n) {
        scanf("%d", &x);
        occ += x;
        if (occ > cap * 0.9)
            critical++;
        i++;
    }
    printf("Final Occupied Beds: %d\n", occ);
    printf("Critical Hours: %d", critical);
    return 0;
}
