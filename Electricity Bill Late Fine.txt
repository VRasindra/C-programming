Electricity Bill Late Fine

#include <stdio.h>

int main() {
    int days, fine = 0;

    scanf("%d", &days);

    if (days > 0) {
        switch (days <= 5) {
            case 1:
                fine = days * 50;
                break;
            default:
                if (days <= 10)
                    fine = days * 100;
                else
                    fine = days * 200;
        }
    }else{

    printf("Fine ₹%d", fine);
   }
    return 0;
}
