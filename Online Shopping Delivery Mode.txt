Online Shopping Delivery Mode
#include <stdio.h>

int main() {
    int mode;

    scanf("%d", &mode);

    if (mode >= 1 && mode <= 3) {
        switch (mode) {
            case 1: 
                printf("Delivery Charge ₹40");
            break;
            case 2:
                printf("Delivery Charge ₹120");
            break;
            case 3: 
               printf("Delivery Charge ₹0");         }
    }

    return 0;
}
