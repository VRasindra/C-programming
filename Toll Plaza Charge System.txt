Toll Plaza Charge System
#include <stdio.h>
int main() {
    int plantype;
    int dataused;
    scanf("%d",&plantype);
    scanf("%f",&dataused);
    switch(plantype){
        case 1:
        if(dataused <=1.0){
            printf("Normal Speed");
        }else{
            printf("speed Reduced");
        }
        break;
        case 2:
        if(dataused<=2.0){
            printf("Normal Speed");
        }else{
            printf("Extra Charges Applied");
        }
    }
    return 0;
}
