3 Department with Maximum Unique Performance Scores
#include <stdio.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    int maxUnique = 0, idx = 0;

    for(int i=0;i<R;i++){
        int unique = 0;
        for(int j=0;j<C;j++){
            int flag = 1;
            for(int k=0;k<j;k++){
                if(a[i][j] == a[i][k]){
                    flag = 0;
                    break;
                }
            }
            if(flag) unique++;
        }
        if(unique > maxUnique){
            maxUnique = unique;
            idx = i;
        }
    }
    printf("%d", idx);
 return 0;
}
