2 Maximum Profit Product Across Branches (2D Array)
#include <stdio.h>
#include <limits.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    int maxProd = INT_MIN;

    for(int i=0;i<R;i++){
        for(int k=i+1;k<R;k++){
            for(int j=0;j<C;j++){
                for(int l=0;l<C;l++){
                    int prod = a[i][j] * a[k][l];
                    if(prod > maxProd)
                        maxProd = prod;
                }
            }
        }
    }
    printf("%d", maxProd);
    return 0;
}
