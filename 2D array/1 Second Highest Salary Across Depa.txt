1 Second Highest Salary Across Departments (2D Array)
#include <stdio.h>
#include <limits.h>

int main() {
    int R, C;
    scanf("%d %d", &R, &C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    int max1 = INT_MIN, max2 = INT_MIN;

    for(int i=0;i<R;i++){
        for(int j=0;j<C;j++){
            if(a[i][j] > max1){
                if(a[i][j] != max1){
                    max2 = max1;
                }
                max1 = a[i][j];
            }
            else if(a[i][j] > max2 && a[i][j] != max1){
                max2 = a[i][j];
            }
        }
    }
    printf("%d", max2);
    return 0;
}
