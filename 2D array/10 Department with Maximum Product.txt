10 Department with Maximum Product Score (2D Array)
#include <stdio.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    long long maxProd = -9e18;
    int idx = 0;

    for(int i=0;i<R;i++){
        long long prod = 1;
        for(int j=0;j<C;j++)
            prod *= a[i][j];

        if(prod > maxProd){
            maxProd = prod;
            idx = i;
        }
    }
    printf("%d", idx);
  return 0;
}
