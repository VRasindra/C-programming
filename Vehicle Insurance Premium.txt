Vehicle Insurance Premium
#include <stdio.h>
int main() {
    int vehicle,age;
    int premium;
    scanf("%d",&vehicle);
    scanf("%d",&age);
    switch(vehicle){
        case 1:
        if(age <=5 ){
            premium=1500;
        }
        else{
           premium=2500;
        }
        break;
        case 2:
        if(age<=5){
            premium=4000;
        }else{
            premium=6000;
        }
    }
    return 0;
}
