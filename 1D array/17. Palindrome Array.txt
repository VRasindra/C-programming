17. Palindrome Array
#include <stdio.h>
int main() {
    int n,a[100],flag=1;
    scanf("%d",&n);
    for(int i=0;i<n;i++) scanf("%d",&a[i]);
    for(int i=0;i<n/2;i++){
        if(a[i]!=a[n-1-i]){
            flag=0; break;
        }
    }
    printf(flag ? "Yes" : "No");
return 0;
}
