16. Best Buy & Sell Profit
#include <stdio.h>
int main() {
    int n,a[100];
    scanf("%d",&n);
    for(int i=0;i<n;i++) scanf("%d",&a[i]);
    int min=a[0],profit=a[1]-a[0];
    for(int i=1;i<n;i++){
        if(a[i]-min > profit)
            profit = a[i]-min;
        if(a[i] < min)
            min = a[i];
    }
    printf("%d",profit);
return 0;
}
