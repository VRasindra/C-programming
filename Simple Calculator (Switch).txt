Simple Calculator (Switch)

include <stdio.h>

int main() {

    int a,b;

    char c;

    scanf("%d %c %d",&a,&b);

    if(c=='+'){

        printf("Result = %d",a+b);

    }else if(c =='-'){

        printf("Result = %d",a-b);

    }else if(c=='*'){

        printf("Result = %d",a*b);

    }else if(c=='%'){

        printf("Result = %d",a/b);

    }else{

      printf("Invalid operator");

    }

    return 0;

}
