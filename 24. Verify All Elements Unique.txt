24. Verify All Elements Unique
#include <stdio.h>
int main() {
    int n,a[100];
    scanf("%d",&n);
    for(int i=0;i<n;i++) scanf("%d",&a[i]);

    for(int i=0;i<n;i++){
        for(int j=i+1;j<n;j++){
            if(a[i]==a[j]){
                printf("No");
            }
        }
    }
    printf("Yes");
return 0;
}
