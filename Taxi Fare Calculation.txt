Taxi Fare Calculation
#include <stdio.h>

int main() {
    int type, distance, fare = 0;

    scanf("%d", &type);
    scanf("%d", &distance);

    if (type >= 1 && type <= 3) {
        switch (type) {
            case 1: 
               fare = distance * 10; 
             break;
            case 2: 
               fare = distance * 15; 
             break;
            case 3:
               fare = distance * 20;        }
    }

    printf("Fare ₹%d", fare);
    return 0;
}
