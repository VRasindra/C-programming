21 Smart Traffic Signal Congestion Controller (While Loop)
#include <stdio.h>
int main() {
    int n, i = 0, x;
    int minutes = 0, streak = 0, maxStreak = 0;

    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &x);
        if (x > 20) {
            minutes++;
            streak++;
            if (streak > maxStreak)
                maxStreak = streak;
        } else {
            streak = 0;
        }
        i++;
    }
    printf("Congestion Minutes: %d\n", minutes);
    printf("Longest Congestion Streak: %d", maxStreak);
    return 0;
}
