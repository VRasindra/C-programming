3: Count Vowels in String Using Pointers ( )
#include <stdio.h>
int main() {
    char str[101];
    char *ptr = str;
    int count = 0;
    fgets(str, sizeof(str), stdin);
    while (*ptr != '\0') {
        if (*ptr == 'a' || *ptr == 'e' || *ptr == 'i' || *ptr == 'o' || *ptr == 'u' ||
            *ptr == 'A' || *ptr == 'E' || *ptr == 'I' || *ptr == 'O' || *ptr == 'U') {
            count++;
        }
        ptr++;
    }
    printf("%d", count);
    return 0;
}
