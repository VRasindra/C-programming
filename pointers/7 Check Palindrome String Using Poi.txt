7: Check Palindrome String Using Pointers
#include <stdio.h>

int main() {
    char str[101];
    char *start, *end;
    int length = 0;
    int isPalindrome = 1;
      scanf("%s", str);
       while (str[length] != '\0') {
        length++;
    }
    start = str;
    end = str + length - 1;
      while (start < end) {
        if (*start != *end) {
            isPalindrome = 0;
            break;
        }
        start++;
        end--;
    }
        if (isPalindrome)
        printf("YES");
    else
        printf("NO");

    return 0;
}
