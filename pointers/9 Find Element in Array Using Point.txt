9: Find Element in Array Using Pointers
#include <stdio.h>
int main() {
    int n, x, position = -1;
    scanf("%d", &n);
    int arr[n];
    int *ptr = arr;
    for (int i = 0; i < n; i++) {
        scanf("%d", ptr + i);
    }
    scanf("%d", &x);
    for (int i = 0; i < n; i++) {
        if (*(ptr + i) == x) {
            position = i + 1;
            break;
        }
    }
    printf("%d", position);
    return 0;
}
