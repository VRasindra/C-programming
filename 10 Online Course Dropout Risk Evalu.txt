10 Online Course Dropout Risk Evaluator (While Loop)
#include <stdio.h>
int main() {
    int n, i = 0, x, inactive = 0;

    scanf("%d", &n);
    while (i < n) {
        scanf("%d", &x);
        if (x == 0)
            inactive++;
        i++;
    }
    printf("Inactive Weeks: %d\n", inactive);
    printf("Risk Status: %s", inactive >= 3 ? "High" : "Low");
    return 0;
}
