18 Maximum Absolute Difference Between Two Departments
#include <stdio.h>
#include <stdlib.h>

int main(){
    int R,C;
    scanf("%d %d",&R,&C);
    int a[R][C];

    for(int i=0;i<R;i++)
        for(int j=0;j<C;j++)
            scanf("%d",&a[i][j]);

    int maxDiff = 0;

    for(int i=0;i<R;i++)
        for(int j=0;j<R;j++)
            if(i!=j)
                for(int x=0;x<C;x++)
                    for(int y=0;y<C;y++){
                        int diff = abs(a[i][x]-a[j][y]);
                        if(diff > maxDiff)
                            maxDiff = diff;
                    }
    printf("%d", maxDiff);
return 0;
}
